<script>
function cycleImage(el) {
  var images = el.getAttribute('data-images').split(',');
  var idx = (parseInt(el.getAttribute('data-idx')) + 1) % images.length;
  el.setAttribute('data-idx', idx);
  el.src = images[idx];
  var dots = el.parentElement.querySelectorAll('.shop-dot');
  for (var i = 0; i < dots.length; i++) {
    dots[i].classList.toggle('shop-dot--active', i === idx);
  }
}
</script>
