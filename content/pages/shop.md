<script>
var lbSource = null;
var lbTouchX = 0;

function setItemImg(img, idx) {
  var images = img.getAttribute('data-images').split(',');
  idx = ((idx % images.length) + images.length) % images.length;
  img.setAttribute('data-idx', idx);
  img.src = images[idx];
  var dots = img.closest('.shop-item').querySelectorAll('.shop-dot');
  for (var i = 0; i < dots.length; i++) {
    dots[i].classList.toggle('shop-dot--active', i === idx);
  }
}

function shiftImg(btn, dir) {
  var img = btn.parentElement.querySelector('img');
  setItemImg(img, parseInt(img.getAttribute('data-idx')) + dir);
}

function openLightbox(img) {
  lbSource = img;
  document.getElementById('lightbox-img').src = img.src;
  document.getElementById('lightbox').classList.add('is-open');
  document.body.style.overflow = 'hidden';
}

function closeLightbox() {
  document.getElementById('lightbox').classList.remove('is-open');
  document.body.style.overflow = '';
  lbSource = null;
}

function lightboxShift(dir) {
  if (!lbSource) return;
  setItemImg(lbSource, parseInt(lbSource.getAttribute('data-idx')) + dir);
  document.getElementById('lightbox-img').src = lbSource.src;
}

document.addEventListener('DOMContentLoaded', function() {
  var lb = document.getElementById('lightbox');

  lb.addEventListener('click', function(e) {
    if (e.target === lb) closeLightbox();
  });

  lb.addEventListener('touchstart', function(e) {
    lbTouchX = e.touches[0].clientX;
  }, { passive: true });

  lb.addEventListener('touchend', function(e) {
    var dx = e.changedTouches[0].clientX - lbTouchX;
    if (Math.abs(dx) > 50) lightboxShift(dx < 0 ? 1 : -1);
  });

  document.addEventListener('keydown', function(e) {
    if (!lb.classList.contains('is-open')) return;
    if (e.key === 'Escape') closeLightbox();
    if (e.key === 'ArrowLeft') lightboxShift(-1);
    if (e.key === 'ArrowRight') lightboxShift(1);
  });
});
</script>
